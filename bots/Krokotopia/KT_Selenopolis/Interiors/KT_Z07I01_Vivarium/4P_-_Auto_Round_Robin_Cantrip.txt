# Priority is P4 plus two alternating players from P1-P3.
# If a scheduled player is low on energy, try the resting player instead.
# plan 1 = P1 P2 P4; plan 2 = P2 P3 P4
# plan 3 = P1 P3 P4; plan 4 = P1 P2 P3; plan 0 = stop.
block Pick_Casters {
    setvar use_p1 = 0
    setvar use_p2 = 0
    setvar use_p3 = 0
    setvar use_p4 = 0
    setvar cast_plan = 0

    if can_p4 == 1 {
        setvar use_p4 = 1
        if rotation == 1 {
            if can_p1 == 1 {
                setvar use_p1 = 1
            }
            if can_p2 == 1 {
                setvar use_p2 = 1
            }
            if use_p1 == 0 {
                if can_p3 == 1 {
                    setvar use_p3 = 1
                }
            }
            if use_p2 == 0 {
                if can_p3 == 1 {
                    setvar use_p3 = 1
                }
            }
        }
        if rotation == 2 {
            if can_p2 == 1 {
                setvar use_p2 = 1
            }
            if can_p3 == 1 {
                setvar use_p3 = 1
            }
            if use_p2 == 0 {
                if can_p1 == 1 {
                    setvar use_p1 = 1
                }
            }
            if use_p3 == 0 {
                if can_p1 == 1 {
                    setvar use_p1 = 1
                }
            }
        }
        if rotation == 3 {
            if can_p1 == 1 {
                setvar use_p1 = 1
            }
            if can_p3 == 1 {
                setvar use_p3 = 1
            }
            if use_p1 == 0 {
                if can_p2 == 1 {
                    setvar use_p2 = 1
                }
            }
            if use_p3 == 0 {
                if can_p2 == 1 {
                    setvar use_p2 = 1
                }
            }
        }

        if use_p1 == 1 {
            if use_p2 == 1 {
                setvar cast_plan = 1
            }
            if use_p3 == 1 {
                setvar cast_plan = 3
            }
        } else {
            if use_p2 == 1 {
                if use_p3 == 1 {
                    setvar cast_plan = 2
                }
            }
        }
    } else {
        if can_p1 == 1 {
            if can_p2 == 1 {
                if can_p3 == 1 {
                    setvar cast_plan = 4
                }
            }
        }
    }
    if cast_plan == 0 {
        print "Not enough energy for three different casters. Stopping."
        kill
    }
}

block Advance_Cast_Rotation {
    if rotation == 1 {
        setvar rotation = 2
    } else {
        if rotation == 2 {
            setvar rotation = 3
        } else {
            setvar rotation = 1
        }
    }
}

# This is deliberately based on target_loc, not an invented safe-spot XYZ.

block Handle_Chest_Found {
    call Read_Energy_Flags
    call Pick_Casters
    print "P1 scouting and anchoring KT chest"
    p1 entitytp KT-Chest-Ritual-002
    p1 waitforfree
    sleep 1

    if cast_plan == 1 {
        print "Round Robin: P1 + P2 + P4 (P3 rests)"
        call P1_Cast_Magic_Touch
        p1 waitforfree
        call P2_From_P1
        call P2_Cast_Magic_Touch
        p2 waitforfree
        call P4_From_P2
        call P4_Cast_Magic_Touch
        p4 waitforfree
    }
    if cast_plan == 2 {
        print "Round Robin: P2 + P3 + P4 (P1 rests)"
        call P2_From_P1
        call P2_Cast_Magic_Touch
        p2 waitforfree
        call P3_From_P2
        call P3_Cast_Magic_Touch
        p3 waitforfree
        call P4_From_P3
        call P4_Cast_Magic_Touch
        p4 waitforfree
    }
    if cast_plan == 3 {
        print "Round Robin: P1 + P3 + P4 (P2 rests)"
        call P1_Cast_Magic_Touch
        p1 waitforfree
        call P3_From_P1
        call P3_Cast_Magic_Touch
        p3 waitforfree
        call P4_From_P3
        call P4_Cast_Magic_Touch
        p4 waitforfree
    }
    if cast_plan == 4 {
        print "P4 low on energy: P1 + P2 + P3"
        call P1_Cast_Magic_Touch
        p1 waitforfree
        call P2_From_P1
        call P2_Cast_Magic_Touch
        p2 waitforfree
        call P3_From_P2
        call P3_Cast_Magic_Touch
        p3 waitforfree
    }

    # Only successful (attempted) chests advance rotation.
    call Advance_Cast_Rotation
    print "Allowing chest to despawn..."
    sleep 6
    # Group stays at the scanning coordinate; next scan moves P1 onward.
    mass tp target_loc
    mass waitforfree
}

block Process_Location {
    # P1 is the scout for all nine chest spawn locations.
    until p1 hasxyz target_loc {
        p1 tp target_loc
    }
    p1 waitforfree
    times 3 {
        if p1 inrange "KT-Chest-Ritual-002" {
            print "KT chest found"
            call Handle_Chest_Found
            break
        } else {
            sleep 1
        }
    }
}
